% bus width mark: slash at #1, number above (h) or right (v)
\def\dpw#1#2{\draw[draw=lsnavy,line width=0.4pt] #1 ++(-0.6,-1.1) -- ++(1.2,2.2);
  \node[dp lbl,anchor=south west] at #1 [xshift=0.3mm,yshift=0.6mm] {#2};}
\def\dpwv#1#2{\draw[draw=lsnavy,line width=0.4pt] #1 ++(-1.1,-0.6) -- ++(2.2,1.2);
  \node[dp lbl,anchor=west] at #1 [xshift=1.2mm] {#2};}
% muxes: vertical (inputs on the left) and horizontal (inputs on top); #1 name, #2 x, #3 y, #4 w, #5 h
\def\dpvmux#1#2#3#4#5{%
  \node[minimum width=#4mm,minimum height=#5mm,inner sep=0pt] (#1) at (#2,#3) {};
  \draw[draw=lsnavy,fill=white,line width=0.6pt,rounded corners=0.8pt] (#1.north west) -- ([yshift=-1.8mm]#1.north east)
    -- ([yshift=1.8mm]#1.south east) -- (#1.south west) -- cycle;}
\def\dphmux#1#2#3#4#5{%
  \node[minimum width=#4mm,minimum height=#5mm,inner sep=0pt] (#1) at (#2,#3) {};
  \draw[draw=lsnavy,fill=white,line width=0.6pt,rounded corners=0.8pt] (#1.north west) -- (#1.north east)
    -- ([xshift=-1.8mm]#1.south east) -- ([xshift=1.8mm]#1.south west) -- cycle;}
